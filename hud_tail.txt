<!-- CLOCK + SESSION -->
<script>
    function getSession() {
        const d = new Date();
        const t = d.getUTCHours() * 60 + d.getUTCMinutes();
        return {
            asian:  t >= 0   && t < 540,
            london: t >= 480 && t < 1020,
            ny:     t >= 780 && t < 1320,
            get afterHours() { return !this.asian && !this.london && !this.ny; }
        };
    }

    function updateClock() {
        const now = new Date();
        let h = now.getHours();
        const m = now.getMinutes().toString().padStart(2, '0');
        const ampm = h >= 12 ? 'PM' : 'AM';
        h = h % 12 || 12;
        document.getElementById('clock-h').textContent = String(h).padStart(2, '0');
        document.getElementById('clock-m').textContent = m;
        document.getElementById('clock-ampm').textContent = ampm;
        updateSession();
    }

    function updateSession() {
        const s = getSession();
        const badge = document.getElementById('session-badge');
        const text  = document.getElementById('session-text');
        const profBadge = document.getElementById('profile-session-badge');
        const asian  = document.getElementById('bar-asian');
        const london = document.getElementById('bar-london');
        const ny     = document.getElementById('bar-ny');

        [asian, london, ny].forEach(el => { el.style.color = '#3f3f46'; });
        badge.className = 'session-badge';

        if (s.ny && s.london) {
            badge.classList.add('session-ny'); text.textContent = 'NY × LONDON'; if(profBadge) profBadge.textContent = 'NY × LONDON';
            london.style.color = '#38bdf8'; ny.style.color = '#4ade80';
        } else if (s.ny) {
            badge.classList.add('session-ny'); text.textContent = 'NEW YORK'; if(profBadge) profBadge.textContent = 'NEW YORK';
            ny.style.color = '#4ade80';
        } else if (s.london) {
            badge.classList.add('session-london'); text.textContent = 'LONDON'; if(profBadge) profBadge.textContent = 'LONDON';
            london.style.color = '#38bdf8';
        } else if (s.asian) {
            badge.classList.add('session-asian'); text.textContent = 'ASIAN'; if(profBadge) profBadge.textContent = 'ASIAN';
            asian.style.color = '#a78bfa';
        } else {
            badge.classList.add('session-after'); text.textContent = 'AFTER HOURS'; if(profBadge) profBadge.textContent = 'AFTER HOURS';
        }
    }

    updateClock();
    setInterval(updateClock, 10000);
</script>

<script>
    // ─────────────────────────────────────────────────────────────
    //  CONFIG: Replace these values with your actual credentials.
    //  YOUTUBE_CHANNEL_ID: Your channel ID (UCxxxxxxxx format)
    //  YOUTUBE_API_KEY:    A YouTube Data API v3 key
    // ─────────────────────────────────────────────────────────────
    const YOUTUBE_CHANNEL_ID = 'REPLACE_WITH_YOUR_CHANNEL_ID';
    const YOUTUBE_API_KEY    = 'REPLACE_WITH_YOUR_YOUTUBE_API_KEY';

    // Check every 3 minutes
    const LIVE_CHECK_INTERVAL = 3 * 60 * 1000;

    let isLive = false;
    let liveMenuOpen = false;

    async function checkYouTubeLive() {
        if (!YOUTUBE_API_KEY || YOUTUBE_API_KEY === 'REPLACE_WITH_YOUR_YOUTUBE_API_KEY') {
            updateBroadcastMsg('⚠ API key not configured.');
            return;
        }
        try {
            const url = `https://www.googleapis.com/youtube/v3/search?part=snippet&channelId=${YOUTUBE_CHANNEL_ID}&eventType=live&type=video&key=${YOUTUBE_API_KEY}`;
            const res = await fetch(url);
            const data = await res.json();
            const live = data.items && data.items.length > 0;
            setLiveState(live, live ? data.items[0].snippet.title : null);
        } catch (err) {
            updateBroadcastMsg('Connection error. Check manually.');
        }
    }

    function setLiveState(live, title = null) {
        isLive = live;
        const btn = document.getElementById('live-btn');
        const dot = document.getElementById('live-dot');
        const label = document.getElementById('live-label');

        dot.className = live ? 'live-dot-red' : 'live-dot-amber';
        btn.className = live ? '' : 'offline';
        label.textContent = live ? '[ BROADCAST_ACTIVE 🔴 ]' : '● TAPPED IN';
        updateBroadcastMsg(live ? `🔴 Live: ${title || 'Stream Active'}` : 'No active broadcast detected.');
    }

    function updateBroadcastMsg(msg) {
        const el = document.getElementById('broadcast-status-msg');
        if (el) el.textContent = msg;
    }

    function toggleLiveMenu() {
        liveMenuOpen = !liveMenuOpen;
        document.getElementById('live-menu').classList.toggle('open', liveMenuOpen);
    }

    document.addEventListener('click', function(e) {
        if (!document.getElementById('live-trigger').contains(e.target)) {
            liveMenuOpen = false;
            document.getElementById('live-menu').classList.remove('open');
        }
    });

    // Initial check + interval
    checkYouTubeLive();
    setInterval(checkYouTubeLive, LIVE_CHECK_INTERVAL);
</script>

    const initAudio = () => {

      const source = audioContext.createMediaElementSource(audioEl);
      source.connect(analyser);
      analyser.connect(audioContext.destination);
      analyser.fftSize = 256;
      const bufferLength = analyser.frequencyBinCount;
      dataArray = new Uint8Array(bufferLength);
      audioEl.play();
      
      // Update UI button

    window.initAudio = initAudio;

    const EngineSphere = () => {
      const meshRef = useRef();
      const materialRef = useRef();
      const { mouse } = useThree();
      const uniforms = useMemo(() => ({
        uTime: { value: 0 },
        uColor: { value: new THREE.Color("#38bdf8") }, // using the blue-cool color from the portfolio
        uAudioLow: { value: 0 },
        uAudioHigh: { value: 0 }
      }), []);

      useFrame((state) => {
        uniforms.uTime.value = state.clock.elapsedTime;
        if (analyser && dataArray) {
          analyser.getByteFrequencyData(dataArray);
          let lowSum = 0; for(let i = 0; i < 10; i++) lowSum += dataArray[i];
          let highSum = 0; for(let i = 80; i < 100; i++) highSum += dataArray[i];
          uniforms.uAudioLow.value += ((lowSum / 10 / 255) - uniforms.uAudioLow.value) * 0.1;
          uniforms.uAudioHigh.value += ((highSum / 20 / 255) - uniforms.uAudioHigh.value) * 0.1;
        }
        if (meshRef.current) {
          const targetRotX = mouse.y * Math.PI;
          const targetRotY = mouse.x * Math.PI;
          meshRef.current.rotation.x += (targetRotX - meshRef.current.rotation.x) * 0.05;
          meshRef.current.rotation.y += (targetRotY - meshRef.current.rotation.y) * 0.05;
          meshRef.current.rotation.y += 0.005 + (uniforms.uAudioHigh.value * 0.05);
          meshRef.current.rotation.z += 0.002 + (uniforms.uAudioHigh.value * 0.02);
        }
      });

      const vertexShader = `
        varying vec2 vUv;
        void main() { vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); }
      `;
      const fragmentShader = `
        uniform float uTime;
        uniform vec3 uColor;
        uniform float uAudioLow;
        varying vec2 vUv;
        void main() {
          vec2 uv = vUv * 60.0;
          vec2 grid = fract(uv);
          float line = step(0.9, grid.x) + step(0.9, grid.y);
          float pulse = (sin(uTime * 3.0) * 0.5 + 0.5);
          float intensePulse = pow(pulse, 2.0);
          float audioReactivity = uAudioLow * 3.0;

        }
      `;
